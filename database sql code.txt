CREATE TABLE `users` (
	`user_id` INT NOT NULL AUTO_INCREMENT,
	`username` VARCHAR(50) NOT NULL COLLATE 'utf8mb4_0900_ai_ci',
	`password` VARCHAR(255) NOT NULL COLLATE 'utf8mb4_0900_ai_ci',
	`role` ENUM('cafe owner','cafe staff') NOT NULL COLLATE 'utf8mb4_0900_ai_ci',
	`created_timestamp` TIMESTAMP NOT NULL DEFAULT (now()),
	PRIMARY KEY (`user_id`) USING BTREE
)
COLLATE='utf8mb4_0900_ai_ci'
ENGINE=InnoDB
;

CREATE TABLE `products` (
	`product_id` INT NOT NULL AUTO_INCREMENT,
	`product_name` VARCHAR(50) NOT NULL COLLATE 'utf8mb4_0900_ai_ci',
	`product_category_id` INT NOT NULL,
	`product_stocks` INT NULL DEFAULT NULL,
	`product_cost` DECIMAL(10,2) NOT NULL,
	`product_selling_price` DECIMAL(10,2) NOT NULL,
	`product_supplier_id` INT NOT NULL,
	`product_type` ENUM('made_to_order','prepared') NOT NULL DEFAULT 'made_to_order' COLLATE 'utf8mb4_0900_ai_ci',
	PRIMARY KEY (`product_id`) USING BTREE,
	INDEX `fk_products_category` (`product_category_id`) USING BTREE,
	INDEX `idx_products_supplier` (`product_supplier_id`) USING BTREE,
	CONSTRAINT `fk_products_category` FOREIGN KEY (`product_category_id`) REFERENCES `product_category` (`product_category_id`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `fk_products_supplier` FOREIGN KEY (`product_supplier_id`) REFERENCES `product_supplier` (`product_supplier_id`) ON UPDATE NO ACTION ON DELETE NO ACTION
)
COLLATE='utf8mb4_0900_ai_ci'
ENGINE=InnoDB
;

CREATE TABLE `product_category` (
	`product_category_id` INT NOT NULL AUTO_INCREMENT,
	`product_category` VARCHAR(50) NOT NULL COLLATE 'utf8mb4_0900_ai_ci',
	PRIMARY KEY (`product_category_id`) USING BTREE
)
COLLATE='utf8mb4_0900_ai_ci'
ENGINE=InnoDB
;

CREATE TABLE `product_ingredients` (
	`product_ingredients_id` INT NOT NULL AUTO_INCREMENT,
	`ingredient_name` VARCHAR(100) NOT NULL COLLATE 'utf8mb4_0900_ai_ci',
	`ingredient_stock` DECIMAL(10,2) NOT NULL DEFAULT '0.00',
	`ingredient_unit` VARCHAR(50) NOT NULL COLLATE 'utf8mb4_0900_ai_ci',
	`ingredient_supplier` VARCHAR(100) NULL DEFAULT NULL COLLATE 'utf8mb4_0900_ai_ci',
	`ingredient_contact` VARCHAR(100) NULL DEFAULT NULL COLLATE 'utf8mb4_0900_ai_ci',
	PRIMARY KEY (`product_ingredients_id`) USING BTREE
)
COLLATE='utf8mb4_0900_ai_ci'
ENGINE=InnoDB
;

CREATE TABLE `product_ingredient_items` (
	`product_ingredient_items_id` INT NOT NULL AUTO_INCREMENT,
	`product_id` INT NOT NULL,
	`product_ingredients_id` INT NOT NULL,
	`quantity` DECIMAL(10,2) NOT NULL DEFAULT '0',
	`unit` VARCHAR(50) NOT NULL DEFAULT '' COLLATE 'utf8mb4_0900_ai_ci',
	PRIMARY KEY (`product_ingredient_items_id`) USING BTREE,
	INDEX `idx_ingredient_items_product` (`product_id`) USING BTREE,
	INDEX `fk_ingredient_items_ingredient` (`product_ingredients_id`) USING BTREE,
	CONSTRAINT `fk_ingredient_items_ingredient` FOREIGN KEY (`product_ingredients_id`) REFERENCES `product_ingredients` (`product_ingredients_id`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `fk_ingredient_items_product` FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON UPDATE NO ACTION ON DELETE NO ACTION
)
COLLATE='utf8mb4_0900_ai_ci'
ENGINE=InnoDB
;

CREATE TABLE `product_supplier` (
	`product_supplier_id` INT NOT NULL AUTO_INCREMENT,
	`supplier_name` VARCHAR(100) NOT NULL COLLATE 'utf8mb4_0900_ai_ci',
	`supplier_contact` VARCHAR(100) NOT NULL COLLATE 'utf8mb4_0900_ai_ci',
	PRIMARY KEY (`product_supplier_id`) USING BTREE
)
COLLATE='utf8mb4_0900_ai_ci'
ENGINE=InnoDB
;

CREATE TABLE `transactions` (
	`transaction_id` INT NOT NULL AUTO_INCREMENT,
	`user_id` INT NOT NULL,
	`transaction_date` DATETIME NOT NULL,
	`transaction_total` DECIMAL(10,2) NOT NULL,
	`transaction_status` ENUM('completed','cancelled') NOT NULL DEFAULT 'completed' COLLATE 'utf8mb4_0900_ai_ci',
	`payment_method` ENUM('cash','online') NOT NULL COLLATE 'utf8mb4_0900_ai_ci',
	`discount` DECIMAL(10,2) NOT NULL DEFAULT '0.00',
	PRIMARY KEY (`transaction_id`) USING BTREE,
	INDEX `fk_transactions_user` (`user_id`) USING BTREE,
	CONSTRAINT `fk_transactions_user` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON UPDATE NO ACTION ON DELETE NO ACTION
)
COLLATE='utf8mb4_0900_ai_ci'
ENGINE=InnoDB
;

CREATE TABLE `transaction_items` (
	`transaction_item_id` INT NOT NULL AUTO_INCREMENT,
	`transaction_id` INT NOT NULL,
	`product_id` INT NOT NULL,
	`quantity` INT NOT NULL,
	`unit_price` DECIMAL(10,2) NOT NULL,
	`subtotal` DECIMAL(10,2) NOT NULL,
	PRIMARY KEY (`transaction_item_id`) USING BTREE,
	INDEX `fk_transaction_id` (`transaction_id`) USING BTREE,
	INDEX `fk_product_id` (`product_id`) USING BTREE,
	CONSTRAINT `fk_product_id` FOREIGN KEY (`product_id`) REFERENCES `products` (`product_id`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `fk_transaction_id` FOREIGN KEY (`transaction_id`) REFERENCES `transactions` (`transaction_id`) ON UPDATE NO ACTION ON DELETE NO ACTION
)
COLLATE='utf8mb4_0900_ai_ci'
ENGINE=InnoDB
;
